/* ---- Helpers de Firestore ---- */
function docRef(id){ return db.collection(COLECCION).doc(id); }

async function leerDoc(id){
  try{
    const snap = await docRef(id).get();
    conectado = true;
    if(!snap.exists) return null;
    const data = snap.data();
    return data && data.json ? JSON.parse(data.json) : null;
  }catch(e){
    conectado = false;
    ultimoErrorStorage = (e && e.message) ? e.message : String(e);
    return null;
  }
}
async function escribirDoc(id, valor){
  try{
    await docRef(id).set({json: JSON.stringify(valor), actualizado: firebase.firestore.FieldValue.serverTimestamp()});
    conectado = true;
    return true;
  }catch(e){
    conectado = false;
    ultimoErrorStorage = (e && e.message) ? e.message : String(e);
    return false;
  }
}

async function guardarProfesores(){ return escribirDoc('profesores', profesores); }

async function cargarDatos(){
  errorCarga = false;
  app.innerHTML = '<div class="loading">Cargando…</div>';

  const urlParams = new URLSearchParams(window.location.search);
  const pIdParam = urlParams.get('p');

  const [p, g] = await Promise.all([leerDoc('profesores'), leerDoc('grupos')]);
  if(p === null && g === null && !conectado){
    errorCarga = true;
    render();
    return;
  }
  profesores = p || [];
  grupos = g || [];

  if(pIdParam){
    const profExiste = profesores.find(prof => prof.id === pIdParam);
    if(profExiste){
      profesorActualId = pIdParam;
      esModoIncrustado = true;
    }
  }

  render();
  iniciarEscuchaProfesoresYGrupos();
}

}

function renderCalendarioGlobal() {
  const cont = document.getElementById('coordinadorContenido');

  let html = `
    <div class="date-row">
      <input type="date" id="inpFechaCoord" value="${fechaSeleccionada}" />

    <div class="roster">

async function renderDirectorioAlumnos() {
  const cont = document.getElementById('coordinadorContenido');
  cont.innerHTML = '<div class="loading">Cargando métricas...</div>';
  
  await cargarTodasAsistencias();

  let html = '';
  const hoyStr = todayISO();

  if(grupos.length === 0) {
      cont.innerHTML = '<div class="empty">No hay grupos creados en la escuela.</div>';
      return;
  }

  grupos.forEach(g => {
      const prof = profesores.find(p => p.id === g.profesorId);
      html += `<div class="day-heading" style="margin-top:16px; border-bottom:1px solid var(--line); padding-bottom:6px;">
                  🎾 ${escapeHtml(g.nombre)} <span style="font-weight:normal; text-transform:none; color:var(--ink-dim);">(${prof?prof.nombre:'-'})</span>
               </div>`;
               
      if(g.alumnos.length === 0) {
          html += `<div class="empty" style="padding:10px 2px;">Grupo sin alumnos</div>`;
      } else {
          g.alumnos.forEach((a, i) => {
            let asistencias = 0;
            let faltas = 0;
            
            const regGrupo = asistenciasCache[g.id] || {};
            // Conteo solo de clases pasadas y actuales
            Object.keys(regGrupo).forEach(fecha => {
                if(fecha <= hoyStr) {
                    const estado = regGrupo[fecha][a.id];
                    if(estado === true) asistencias++;
                    if(estado === false) faltas++;
                }
            });

            html += `
              <div class="roster-row" style="padding:12px 2px; cursor:pointer;" onclick="abrirCalendarioAlumno('${a.id}', '${g.id}')">
                <span class="num" style="width:16px;">${i+1}</span>
                <div class="main">
                  <div class="name" style="font-size:15px;">${escapeHtml(a.nombre)}</div>
                  <div class="meta" style="font-size:11px;">Ver historial de calendario</div>
                </div>
                <div style="font-size:13px; font-weight:bold; display:flex; gap:12px; text-align:right;">
                    <span style="color:var(--presente);" title="Asistencias">✓ ${asistencias}</span>
                    <span style="color:var(--ausente);" title="Faltas">✕ ${faltas}</span>
                </div>
              </div>
            `;
          });
      }
  });
  
  cont.innerHTML = html;
}

function copiarEnlaceProfesor(idProfesor){
  const urlBase = window.location.href.split('?')[0];
  const enlace = `${urlBase}?p=${encodeURIComponent(idProfesor)}`;
  navigator.clipboard.writeText(enlace).then(()=>{
    alert('¡Enlace copiado al portapapeles! Puedes enviárselo al profesor.');
  }).catch(()=>{
    prompt('Copia este enlace para el profesor:', enlace);
  });
}

function historialDots(grupoId, alumnoId){
  const datos = asistenciasCache[grupoId] || {};
  const fechas = Object.keys(datos).sort().reverse().slice(0,6).reverse();
  if(fechas.length === 0) return '<span class="dot"></span><span class="dot"></span><span class="dot"></span>';
  return fechas.map(f=>{
    const val = datos[f] ? datos[f][alumnoId] : undefined;
    const cls = val === true ? 'si' : (val === false ? 'no' : '');
    return `<span class="dot ${cls}"></span>`;
  }).join('');
}

function renderAlumnos(){
  const grupo = grupos.find(g=>g.id===grupoActualId);
  if(!grupo) return;
  const cont = document.getElementById('listaAlumnos');
  if(!cont) return;
  const registro = (asistenciasCache[grupo.id] && asistenciasCache[grupo.id][fechaSeleccionada]) || {};

  cont.innerHTML = '';
  if(grupo.alumnos.length===0){
    cont.innerHTML = '<div class="empty">No hay alumnos en este grupo.</div>';
    return;
  }

  grupo.alumnos.forEach((a,i)=>{
    const estado = registro[a.id];
    let textoBtn = 'Sin marcar';
    let claseBtn = 'sin-marcar';

    if(estado === true){ textoBtn = 'Presente'; claseBtn = 'presente'; }
    else if(estado === false){ textoBtn = 'Ausente'; claseBtn = 'ausente'; }

    const row = document.createElement('div');
    row.className = 'lineup-row';
    row.innerHTML = `
      <span class="num">${String(i+1).padStart(2,'0')}</span>
      <div class="info" title="Ver calendario">
        <div class="sname">${escapeHtml(a.nombre)}</div>
        <div class="dots">${historialDots(grupo.id, a.id)}</div>
      </div>
      <button class="state-btn ${claseBtn}">${textoBtn}</button>
      <button class="icon-btn" title="Editar Nombre" onclick="abrirDialogoAlumno('${a.id}')">✎</button>
      <button class="remove-x" title="Quitar alumno del grupo">✕</button>
    `;
    row.querySelector('.info').onclick = ()=> abrirCalendarioAlumno(a.id, grupo.id);

    row.querySelector('.state-btn').onclick = ()=>{
      if(!asistenciasCache[grupo.id]) asistenciasCache[grupo.id] = {};
      if(!asistenciasCache[grupo.id][fechaSeleccionada]) asistenciasCache[grupo.id][fechaSeleccionada] = {};
      const actual = asistenciasCache[grupo.id][fechaSeleccionada][a.id];

      if(actual === undefined) asistenciasCache[grupo.id][fechaSeleccionada][a.id] = true;
      else if(actual === true) asistenciasCache[grupo.id][fechaSeleccionada][a.id] = false;
      else delete asistenciasCache[grupo.id][fechaSeleccionada][a.id];

      sucioAsistencia = true;
      renderAlumnos();
      document.getElementById('saveNote').textContent = 'Cambios sin guardar';
      document.getElementById('saveNote').style.color = 'var(--ink-dim)';
    };
    row.querySelector('.remove-x').onclick = ()=> quitarAlumno(a.id);
    cont.appendChild(row);
  });
}

  const month = calFecha.getMonth();
  document.getElementById('mesCalendarioLabel').textContent = calFecha.toLocaleDateString('es-ES', {month:'long', year:'numeric'});

  const primerDiaMes = new Date(year, month, 1).getDay();
  const offsetLunes = (primerDiaMes + 6) % 7;
  const diasEnMes = new Date(year, month+1, 0).getDate();
  const datos = asistenciasCache[calGrupoId] || {};
  
  const idxDiaClase = DIAS_SEMANA.indexOf(grupo ? grupo.dia : null); // 0 = Domingo, 1 = Lunes...
  const hoyISO = todayISO();

  let html = '';
  for(let i=0; i<offsetLunes; i++) html += '<div class="cal-cell vacia"></div>';
  
  for(let d=1; d<=diasEnMes; d++){
    const iso = `${year}-${String(month+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
    const reg = datos[iso];
    let cls = 'cal-cell';
    
    // Verificamos si este día de la semana coincide con el día del grupo
    const esDiaDeClase = new Date(year, month, d).getDay() === idxDiaClase;

    if(reg && calAlumnoId in reg && reg[calAlumnoId] !== undefined) {
       cls += reg[calAlumnoId] ? ' si' : ' no';
    } else if(esDiaDeClase && iso > hoyISO) {
       // Día de clase en el futuro (neutro/gris)
       cls += ' futura';
    }
    
    html += `<div class="${cls}">${d}</div>`;
  }
  document.getElementById('gridCalendario').innerHTML = html;
}
